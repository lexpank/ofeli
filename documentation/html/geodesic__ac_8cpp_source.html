92 int GeodesicAC::compute_external_speed_Fd(int offset) {
93 int x, y;
94 find_xy_position(offset,x,y); // x and y passed by reference
95
96 int sum_local_out, n_local_out, sum_local_in, n_local_in;
97
98 // if high gradient value
99 if( double(img_data[offset]) > 0.7*double(otsu_threshold1) ) {
100 sum_local_out = 0;
101 n_local_out = 0;
102 sum_local_in = 0;
103 n_local_in = 0;
104
105 if( y > 0 ) {
106 if( phi[ find_offset(x,y-1) ] < 0 ) {
107 sum_local_in += int( img_data[ find_offset(x,y-1) ] );
108 n_local_in++;
109 } else {
110 sum_local_out += int( img_data[ find_offset(x,y-1) ] );
111 n_local_out++;
112 }
113 }
114 if( x > 0 ) {
115 if( phi[ find_offset(x-1,y) ] < 0 ) {
116 sum_local_in += int( img_data[ find_offset(x-1,y) ] );
117 n_local_in++;
118 } else {
119 sum_local_out += int( img_data[ find_offset(x-1,y) ] );
120 n_local_out++;
121 }
122 }
123 if( x < img_width-1 ) {
124 if( phi[ find_offset(x+1,y) ] < 0 ) {
125 sum_local_in += int( img_data[ find_offset(x+1,y) ] );
126 n_local_in++;
127 } else {
128 sum_local_out += int( img_data[ find_offset(x+1,y) ] );
129 n_local_out++;
130 }
131 }
132 if( y < img_height-1 ) {
133 if( phi[ find_offset(x,y+1) ] < 0 ) {
134 sum_local_in += int( img_data[ find_offset(x,y+1) ] );
135 n_local_in++;
136 } else {
137 sum_local_out += int( img_data[ find_offset(x,y+1) ] );
138 n_local_out++;
139 }
140 }
141
142 return sum_local_out*n_local_in - sum_local_in*n_local_out;
143 } else { // if low gradient value
144 if( isInside ) { // if high gradient values are inside the curve
145 return -1; // inward movement
146 } else {
147 return 1; // outward movement
148 }
149 }
150 }

153 unsigned char GeodesicAC::otsu_method(const unsigned char* img_data1, int img_size1) {
154 int histogram[256];
155
156 for( int I = 0; I < 256; I++ ) {
157 histogram[I] = 0;
158 }
159
160 for( int offset = 0; offset < img_size1; offset++ ) {
161 histogram[ img_data1[offset] ]++;
162 }
163
164 int sum = 0;
165 for( int I = 0; I < 256; I++ ) {
166 sum += I*histogram[I];
167 }
168
169 int weight1, weight2, sum1, threshold;
170 double mean1, mean2, var_t, var_max;
171
172 threshold = 127; // value returned in the case of an totally homogeneous image
173
174 weight1 = 0;
175 sum1 = 0;
176 var_max = -1.0;
177
178 // 256 values ==> 255 thresholds t evaluated
179 // class1 <= t and class2 > t
180 for( int t = 0; t < 255; t++ ) {
181 weight1 += histogram[t];
182 if( weight1 == 0 ) {
183 continue;
184 }
185
186 weight2 = img_size1-weight1;
187 if( weight2 == 0 ) {
188 break;
189 }
190
191 sum1 += t*histogram[t];
192
193 mean1 = sum1/weight1;
194 mean2 = (sum-sum1)/weight2; // sum2 = sum-sum1
195
196 var_t = double(weight1)*double(weight2)*(mean1-mean2)*(mean1-mean2);
197
198 if( var_t > var_max ) {
199 var_max = var_t;
200 threshold = t;
201 }
202 }
203
204 return (unsigned char)(threshold);
205 }

207 bool GeodesicAC::find_direction_evolution() {
208 int sum_in = 0;
209 int sum_out = 0;
210
211 for( int offset = 0; offset < img_size; offset++ ) {
212 if( phi[offset] < 0 ) {
213 sum_in += int(img_data[offset]);
214 } else {
215 sum_out += int(img_data[offset]);
216 }
217 }
218
219 if( sum_in > sum_out ) {
220 return true;
221 } else {
222 return false;
223 }
224 }
