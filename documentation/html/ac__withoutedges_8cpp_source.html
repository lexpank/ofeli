101void ACwithoutEdges::initialize_sums()
102{
103 sum_in = 0;
104 sum_out = 0;
105 n_in = 0;
106 n_out = 0;
107
108 for( int offset = 0; offset < img_size; offset++ )
109 {
110 if( phi[offset] > 0 )
111 {
112 sum_out += int(img_data[offset]);
113 n_out++;
114 }
115 else
116 {
117 sum_in += int(img_data[offset]);
118 n_in++;
119 }
120 }
121
122 return;
123}
124
125void ACwithoutEdges::calculate_means()
126{
127 if( n_out != 0 )
128 {
129 Cout = sum_out/n_out;
130 }
131
132 if( n_in != 0 )
133 {
134 Cin = sum_in/n_in;
135 }
136
137 return;
138}

140int ACwithoutEdges::compute_external_speed_Fd(int offset)
141{
142 I = int(img_data[offset]); // intensity of the current pixel
143
144 return lambda_out*square(I-Cout) - lambda_in*square(I-Cin);
145}
