110#include "imageviewer.hpp"
111
112int main(int argc, char *argv[])
113{
114 // to have the windows theme
115 //QApplication::setStyle(new QWindowsStyle);
116
117 QApplication app(argc, argv);
118 QCoreApplication::setApplicationName(QStringLiteral("Ofeli"));
119 QCoreApplication::setOrganizationName(QStringLiteral("Bessy"));
120 QCoreApplication::setApplicationVersion(QStringLiteral(OFELI_VERSION));
121
122 QSettings settings("Bessy", "Ofeli");
123 int language = settings.value("language",0).toInt();
124
125 QTranslator translator_qt_widget;
126 QTranslator translator_ofeli;
127 QString locale;
128
129 switch( language )
130 {
131 case 0 :
132 {
133 locale = QLocale::system().name().section('_', 0, 0);
134 break;
135 }
136 case 1 :
137 {
138 locale = QString("en");
139 break;
140 }
141 case 2 :
142 {
143 locale = QString("fr");
144 break;
145 }
146 default :
147 {
148 locale = QString("en");
149 }
150 }
151
152 if( translator_qt_widget.load( QString("qt_") + locale, QLibraryInfo::path(QLibraryInfo::TranslationsPath) ) )
153 {
154 app.installTranslator(&translator_qt_widget);
155 }
156
157 if( translator_ofeli.load( QString(":/i18n/Ofeli_") + locale + QStringLiteral(".qm") ) )
158 {
159 app.installTranslator(&translator_ofeli);
160 }
161
162 ofeli::ImageViewer imageViewer;
163 imageViewer.show();
164
165 return app.exec();
166}
