106void ACwithoutEdgesYUV::initialize_for_each_frame()
107{
108 int it = get_iteration();
109
110 ActiveContour::initialize_for_each_frame();
111
112 // condition for video tracking :
113 // If the active contour moves enough ( >120 ), variables sums, n_in, n_out
114 // are too different and are recalculated.
115 // If the active contour does not evolve ( < 5 because a last cycle 2 is performed )
116 // in the case of uniform images (or during the object initialization),
117 // variables are (re)calculated for convergence.
118 // Otherwise, variables are not calculated.
119 if( it > 120 || it < 5 )
120 {
121 initialize_sums();
122 }
123 calculate_means();
124
125 return;
126}

128void ACwithoutEdgesYUV::initialize_sums()
129{
130 sum_in_R = 0;
131 sum_out_R = 0;
132
133 sum_in_G = 0;
134 sum_out_G = 0;
135
136 sum_in_B = 0;
137 sum_out_B = 0;
138
139 n_in = 0;
140 n_out = 0;
141
142 for( int offset = 0; offset < img_size; offset++ )
143 {
144 RGB_from_buffer(offset);
145
146 if( phi[offset] > 0 )
147 {
148 sum_out_R += R;
149 sum_out_G += G;
150 sum_out_B += B;
151
152 n_out++;
153 }
154 else
155 {
156 sum_in_R += R;
157 sum_in_G += G;
158 sum_in_B += B;
159
160 n_in++;
161 }
162 }
163
164 return;
165}

167void ACwithoutEdgesYUV::calculate_means()
168{
169 if( n_out != 0 )
170 {
171 CoutR = sum_out_R/n_out;
172 CoutG = sum_out_G/n_out;
173 CoutB = sum_out_B/n_out;
174 }
175
176 if( n_in != 0 )
177 {
178 CinR = sum_in_R/n_in;
179 CinG = sum_in_G/n_in;
180 CinB = sum_in_B/n_in;
181 }
182
183 calculate_YUV(CoutR, CoutG, CoutB, CoutYUV);
184 calculate_YUV(CinR, CinG, CinB, CinYUV);
185
186 return;
187}

189int ACwithoutEdgesYUV::compute_external_speed_Fd(int offset)
190{
191 RGB_from_buffer(offset);
192
193 int YUV[3];
194 calculate_YUV(R, G, B, YUV);
195
196 return lambda_out*( alpha*square(YUV[0]-CoutYUV[0])
197 + beta*square(YUV[1]-CoutYUV[1])
198 + gamma*square(YUV[2]-CoutYUV[2]) )
199
200 - lambda_in*( alpha*square(YUV[0]-CinYUV[0])
201 + beta*square(YUV[1]-CinYUV[1])
202 + gamma*square(YUV[2]-CinYUV[2]) );
203}

205void ACwithoutEdgesYUV::updates_for_means_in1()
206{
207 sum_out_R -= R;
208 sum_out_G -= G;
209 sum_out_B -= B;
210 n_out--;
211
212 sum_in_R += R;
213 sum_in_G += G;
214 sum_in_B += B;
215 n_in++;
216
217 return;
218}
