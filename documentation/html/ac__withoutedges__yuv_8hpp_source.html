188inline void ACwithoutEdgesYUV::RGB_from_buffer(int offset)
189{
190 #ifdef RGB_PLANAR_IMAGE_DATA_BUFFER
191
192 // RGB planar image data buffer, with R1 R2 R3 ... G1 G2 G3 ... B1 B2 B3 ... (for Matlab)
193 R = int(img_data[ offset ]);
194 G = int(img_data[ offset+img_size ]);
195 B = int(img_data[ offset+2*img_size ]);
196
197 #else
198
199 // RGB interleaved image data buffer, with R1 G1 B1 R2 G2 B2 ...
200 R = int(img_data[ 3*offset ]);
201 G = int(img_data[ 3*offset+1 ]);
202 B = int(img_data[ 3*offset+2 ]);
203
204 #endif
205
206 return;
207}

209inline void ACwithoutEdgesYUV::calculate_YUV(int R, int G, int B, int YUV[])
210{
211 YUV[0] = ( ( 66 * R + 129 * G + 25 * B + 128) >> 8) + 16; // Y
212 YUV[1] = ( ( -38 * R - 74 * G + 112 * B + 128) >> 8) + 128; // U
213 YUV[2] = ( ( 112 * R - 94 * G - 18 * B + 128) >> 8) + 128; // V
214
215 return;
216}
